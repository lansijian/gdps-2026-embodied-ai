cmake_minimum_required(VERSION 3.5)
project(g1_nav_tf_authority)
set(CMAKE_CXX_STANDARD 14)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(tf_authority src/tf_authority.cpp)
ament_target_dependencies(tf_authority rclcpp tf2_msgs std_msgs)
install(TARGETS tf_authority DESTINATION lib/${PROJECT_NAME})
ament_package()
