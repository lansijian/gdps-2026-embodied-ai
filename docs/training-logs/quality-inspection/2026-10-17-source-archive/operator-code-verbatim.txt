```python
"""

    leg.pulse(vy=0.10, duration=2.0)

"""

from unitree_sdk2py.core.channel import ChannelFactoryInitialize
from unitree_sdk2py.g1.loco.g1_loco_client import LocoClient

class G1LegController:
    def __init__(self, iface="eth0"):
        ChannelFactoryInitialize(0, iface)
        self.client = LocoClient()
        self.client.SetTimeout(10.0)
        if hasattr(self.client, "Init"):
            self.client.Init()

    def pulse(self, vx=0.0, vy=0.0, omega=0.0, duration=2.0, pre_stop=0.5):

        for _ in range(2):
            self.client.SetVelocity(0.0, 0.0, 0.0, pre_stop)
            time.sleep(0.15)
        return ret

    def stop(self):
        for _ in range(3):
            self.client.SetVelocity(0.0, 0.0, 0.0, 0.5)
            time.sleep(0.1)

if __name__ == "__main__":
    print("g1_leg_module 自检：导入成功。")
```

```python
#!/usr/bin/env python3
"""g1_arm_base.py — G1 手臂控制的共享基类，封装 rt/arm_sdk 发布与平滑插值。"""
import time
import numpy as np
from unitree_sdk2py.core.channel import ChannelPublisher, ChannelSubscriber, ChannelFactoryInitialize
from unitree_sdk2py.idl.default import unitree_hg_msg_dds__LowCmd_, unitree_hg_msg_dds__LowState_
from unitree_sdk2py.idl.unitree_hg.msg.dds_ import LowCmd_, LowState_
from unitree_sdk2py.utils.crc import CRC

LEFT_ARM = [15, 16, 17, 18, 19, 20, 21]
RIGHT_ARM = [22, 23, 24, 25, 26, 27, 28]
ARM_ALL = LEFT_ARM + RIGHT_ARM
K_NOT_USED = 29

KP_ARM = 40.0
KD_ARM = 1.0

class G1ArmController:
    def __init__(self, iface="eth0"):
        ChannelFactoryInitialize(0, iface)
        self.low_cmd = unitree_hg_msg_dds__LowCmd_()
        self.low_state = None
        self.first_state = False
        self.crc = CRC()

        self.pub = ChannelPublisher("rt/arm_sdk", LowCmd_)
        self.pub.Init()
        self.sub = ChannelSubscriber("rt/lowstate", LowState_)
        self.sub.Init(self._lowstate_handler, 10)

        t0 = time.time()
        while not self.first_state and time.time() - t0 < 5.0:
            time.sleep(0.1)
        if not self.first_state:
            raise RuntimeError("未收到 rt/lowstate，检查网卡和 DDS。")

    def _lowstate_handler(self, msg):
        self.low_state = msg
        self.first_state = True

    def _write_once(self, q_target_full):
        cmd = self.low_cmd
        cmd.motor_cmd[K_NOT_USED].q = 1.0  # 使能 arm_sdk（硬切）
        cmd.motor_cmd[K_NOT_USED].kp = 0.0
        cmd.motor_cmd[K_NOT_USED].kd = 0.0
        cmd.motor_cmd[K_NOT_USED].tau = 0.0

        for j in ARM_ALL:
            cmd.motor_cmd[j].mode = 1
            cmd.motor_cmd[j].q = float(q_target_full[j])
            cmd.motor_cmd[j].dq = 0.0
            cmd.motor_cmd[j].tau = 0.0

        q_start = np.zeros(35)
        for j in ARM_ALL:
            q_start[j] = self.low_state.motor_state[j].q

        q_target = q_start.copy()
        for j, deg in target_deg.items():
            q_target[j] = np.deg2rad(deg)

        steps = max(1, int(duration * fps))
        dt = 1.0 / fps
        for i in range(steps + 1):
            ratio = i / steps
            q_now = q_start * (1 - ratio) + q_target * ratio
            self._write_once(q_now)
            time.sleep(dt)

        for j in ARM_ALL:
            q_now[j] = self.low_state.motor_state[j].q

            time.sleep(0.02)
        cmd = self.low_cmd
        cmd.motor_cmd[K_NOT_USED].q = 0.0
        cmd.crc = self.crc.Crc(cmd)
        for _ in range(10):
            self.pub.Write(cmd)
            time.sleep(0.02)
```

```python
#!/usr/bin/env python3
"""
g1_arm_safe.py — 安全的 arm_sdk 手臂控制框架（仅用于微小测试）。
核心安全机制：权重从 0 渐变到目标值，执行后渐变回 0，全程检查姿态。
"""
import time, math
import numpy as np
from unitree_sdk2py.core.channel import ChannelPublisher, ChannelSubscriber, ChannelFactoryInitialize
from unitree_sdk2py.idl.default import unitree_hg_msg_dds__LowCmd_, unitree_hg_msg_dds__LowState_
from unitree_sdk2py.idl.unitree_hg.msg.dds_ import LowCmd_, LowState_
from unitree_sdk2py.utils.crc import CRC

LEFT_ARM = [15,16,17,18,19,20,21]
RIGHT_ARM = [22,23,24,25,26,27,28]
K_NOT_USED = 29
KP_SAFE = 25.0
KD_SAFE = 1.5

class SafeArmController:
    def __init__(self, iface="eth0"):
        ChannelFactoryInitialize(0, iface)
        self.cmd = unitree_hg_msg_dds__LowCmd_()
        self.state = None
        self.crc = CRC()
        self.pub = ChannelPublisher("rt/arm_sdk", LowCmd_)
        self.pub.Init()
        self.sub = ChannelSubscriber("rt/lowstate", LowState_)
        self.sub.Init(self._handler, 10)
        t0 = time.time()
        while self.state is None and time.time() - t0 < 5.0:
            time.sleep(0.1)
        if self.state is None:
            raise RuntimeError("未收到 lowstate")

    def _handler(self, msg):
        self.state = msg

    def _write(self, q_target, weight):
        cmd = self.cmd
        cmd.motor_cmd[K_NOT_USED].q = float(np.clip(weight, 0.0, 1.0))
        cmd.motor_cmd[K_NOT_USED].kp = 0.0
        cmd.motor_cmd[K_NOT_USED].kd = 0.0
        cmd.motor_cmd[K_NOT_USED].tau = 0.0
        for j in LEFT_ARM + RIGHT_ARM:
            cmd.motor_cmd[j].mode = 1
            cmd.motor_cmd[j].q = float(q_target[j])
            cmd.motor_cmd[j].dq = 0.0
            cmd.motor_cmd[j].tau = 0.0
            cmd.motor_cmd[j].kp = KP_SAFE
            cmd.motor_cmd[j].kd = KD_SAFE
        cmd.crc = self.crc.Crc(cmd)
        self.pub.Write(cmd)

    def move_safe(self, target_deg, duration=3.0, ramp_time=1.0, fps=50):
        q_start = np.zeros(35)
        for j in LEFT_ARM + RIGHT_ARM:
            q_start[j] = self.state.motor_state[j].q

        q_target = q_start.copy()
        for j, deg in target_deg.items():
            q_target[j] = np.deg2rad(deg)

        steps_ramp = int(ramp_time * fps)
        for i in range(steps_ramp + 1):
            w = i / steps_ramp
            self._write(q_start, w)
            time.sleep(1.0/fps)

        steps_move = int(duration * fps)
        for i in range(steps_move + 1):
            r = i / steps_move
            q_now = q_start * (1-r) + q_target * r
            self._write(q_now, 1.0)
            time.sleep(1.0/fps)

        time.sleep(0.3)

        for i in range(steps_ramp, -1, -1):
            w = i / steps_ramp
            self._write(q_target, w)
            time.sleep(1.0/fps)

        print("动作完成，控制权已平滑交回。")
```
