import time, math
import numpy as np
from unitree_sdk2py.core.channel import ChannelPublisher, ChannelSubscriber, ChannelFactoryInitialize
from unitree_sdk2py.idl.default import unitree_hg_msg_dds__LowCmd_
from unitree_sdk2py.idl.unitree_hg.msg.dds_ import LowCmd_, LowState_
from unitree_sdk2py.utils.crc import CRC

LEFT_SHOULDER, RIGHT_SHOULDER = 15, 22
DELTA_DEG = -60.0
MAX_WEIGHT = 1.0
DURATION = 4.0
FPS = 50

WAIST_LOCK = [12, 13, 14]
LEFT_OTHER  = [16, 17, 18, 19, 20, 21]
RIGHT_OTHER = [23, 24, 25, 26, 27, 28]

ChannelFactoryInitialize(0, "eth0")
state = {"msg": None}
sub = ChannelSubscriber("rt/lowstate", LowState_)
sub.Init(lambda m: state.__setitem__("msg", m), 10)
t0 = time.time()
while state["msg"] is None and time.time() - t0 < 5.0:
    time.sleep(0.05)
if state["msg"] is None:
    raise RuntimeError("未收到 lowstate")
msg0 = state["msg"]

ql0 = msg0.motor_state[LEFT_SHOULDER].q
ql_t = ql0 + np.deg2rad(DELTA_DEG)
qr0 = msg0.motor_state[RIGHT_SHOULDER].q
qr_t = qr0 + np.deg2rad(DELTA_DEG)
waist_q = {j: msg0.motor_state[j].q for j in WAIST_LOCK}
left_keep  = {j: msg0.motor_state[j].q for j in LEFT_OTHER}
right_keep = {j: msg0.motor_state[j].q for j in RIGHT_OTHER}

pub = ChannelPublisher("rt/arm_sdk", LowCmd_)
pub.Init()
crc = CRC()
cmd = unitree_hg_msg_dds__LowCmd_()

def tilt(m):
    qw, qx, qy, qz = m.imu_state.quaternion
    return math.acos(max(-1, min(1, 1 - 2*(qx*qx + qy*qy))))

def write(ql, qr, weight):
    m = state["msg"]
    if m is None or tilt(m) > 0.7:
        return False
    cmd.motor_cmd[29].q = float(np.clip(weight, 0, MAX_WEIGHT))
    cmd.motor_cmd[29].kp = 0; cmd.motor_cmd[29].kd = 0; cmd.motor_cmd[29].tau = 0
    for j in WAIST_LOCK:
        cmd.motor_cmd[j].mode = 1
        cmd.motor_cmd[j].q = waist_q[j]
        cmd.motor_cmd[j].dq = 0; cmd.motor_cmd[j].tau = 0
        cmd.motor_cmd[j].kp = 50; cmd.motor_cmd[j].kd = 1.5
    cmd.motor_cmd[LEFT_SHOULDER].mode = 1
    cmd.motor_cmd[LEFT_SHOULDER].q = float(ql)
    cmd.motor_cmd[LEFT_SHOULDER].dq = 0; cmd.motor_cmd[LEFT_SHOULDER].tau = 0
    cmd.motor_cmd[LEFT_SHOULDER].kp = 35; cmd.motor_cmd[LEFT_SHOULDER].kd = 1.5
    cmd.motor_cmd[RIGHT_SHOULDER].mode = 1
    cmd.motor_cmd[RIGHT_SHOULDER].q = float(qr)
    cmd.motor_cmd[RIGHT_SHOULDER].dq = 0; cmd.motor_cmd[RIGHT_SHOULDER].tau = 0
    cmd.motor_cmd[RIGHT_SHOULDER].kp = 35; cmd.motor_cmd[RIGHT_SHOULDER].kd = 1.5
    for j in LEFT_OTHER:
        cmd.motor_cmd[j].mode = 1
        cmd.motor_cmd[j].q = left_keep[j]
        cmd.motor_cmd[j].dq = 0; cmd.motor_cmd[j].tau = 0
        cmd.motor_cmd[j].kp = 15; cmd.motor_cmd[j].kd = 1
    for j in RIGHT_OTHER:
        cmd.motor_cmd[j].mode = 1
        cmd.motor_cmd[j].q = right_keep[j]
        cmd.motor_cmd[j].dq = 0; cmd.motor_cmd[j].tau = 0
        cmd.motor_cmd[j].kp = 15; cmd.motor_cmd[j].kd = 1
    cmd.crc = crc.Crc(cmd)
    pub.Write(cmd)
    return True

def ramp(ql_a, ql_b, qr_a, qr_b, wa, wb, sec):
    n = max(1, int(sec * FPS))
    for i in range(n + 1):
        r = 0.5 - 0.5 * math.cos(math.pi * i / n)
        if not write(ql_a + (ql_b - ql_a) * r, qr_a + (qr_b - qr_a) * r, wa + (wb - wa) * r):
            print("⚠️ 安全中止"); return False
        time.sleep(1 / FPS)
    return True

print("\n5 秒后开始。双臂 -60°，腰部 kp=50。")
for i in (5, 4, 3, 2, 1):
    print(f"  {i}..."); time.sleep(1)

try:
    ramp(ql0, ql0, qr0, qr0, 0, MAX_WEIGHT, 2.0)
    ramp(ql0, ql_t, qr0, qr_t, MAX_WEIGHT, MAX_WEIGHT, DURATION)
    time.sleep(0.8)
    ramp(ql_t, ql0, qr_t, qr0, MAX_WEIGHT, MAX_WEIGHT, DURATION)
    ramp(ql0, ql0, qr0, qr0, MAX_WEIGHT, 0, 2.0)
    print("✅ 完成。")
except Exception as e:
    print(f"❌ 中止: {e}")
    for _ in range(20):
        write(ql0, qr0, 0.0); time.sleep(0.02)

pub.Close()
sub.Close()
