#!/usr/bin/env python3
"""g1_left.py — 安全向左走一小段距离。默认 vy=0.10 m/s, 2s ≈ 0.2m
注意: vy 正方向若实际为右, 把默认值改为 -0.10 即可。"""
import time, argparse
from unitree_sdk2py.core.channel import ChannelFactoryInitialize
from unitree_sdk2py.g1.loco.g1_loco_client import LocoClient

def main():
    p = argparse.ArgumentParser()
    p.add_argument("--iface", default="eth0")
    p.add_argument("--vy", type=float, default=0.10, help="侧移速度 m/s")
    p.add_argument("--duration", type=float, default=2.0)
    a = p.parse_args()

    ChannelFactoryInitialize(0, a.iface)
    c = LocoClient(); c.SetTimeout(10.0)
    c.Init() if hasattr(c, "Init") else None

    dist = abs(a.vy) * a.duration
    print(f"[LEFT] vy={a.vy} m/s, {a.duration}s, 预计距离≈{dist:.2f}m")
    print("3秒后开始，请确认急停在手、左侧清空...")
    for i in (3, 2, 1):
        print(f"  {i}..."); time.sleep(1)

    ret = c.SetVelocity(0.0, a.vy, 0.0, a.duration)
    print(f"SetVelocity 返回: {ret}")
    time.sleep(a.duration + 0.3)
    for _ in range(2):
        c.SetVelocity(0.0, 0.0, 0.0, 0.5); time.sleep(0.3)
    print("[LEFT] 完成。")

if __name__ == "__main__":
    main()
