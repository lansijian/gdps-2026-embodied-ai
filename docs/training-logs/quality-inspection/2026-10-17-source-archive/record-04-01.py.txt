#!/usr/bin/env python3
"""
move_pulse.py — 在 FSM 500 + balance_mode=1 下发送单次微小速度脉冲。
默认 vx=0.1 m/s, duration=0.5s。支持 --vx/--vy/--omega/--duration 参数。
用法: python3 move_pulse.py [--iface eth0] [--vx 0.1] [--duration 0.5]
"""
import sys
import time
import argparse

from unitree_sdk2py.core.channel import ChannelFactoryInitialize
from unitree_sdk2py.g1.loco.g1_loco_client import LocoClient

def confirm(prompt):
    ans = input(f"{prompt} [yes/NO]: ").strip().lower()
    if ans != "yes":
        print("操作已取消。")
        sys.exit(0)

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--iface", default="eth0")
    parser.add_argument("--vx", type=float, default=0.1, help="前进速度 m/s (负值后退)")
    parser.add_argument("--vy", type=float, default=0.0, help="侧移速度 m/s")
    parser.add_argument("--omega", type=float, default=0.0, help="偏航角速度 rad/s")
    parser.add_argument("--duration", type=float, default=0.5, help="速度持续时间 s")
    args = parser.parse_args()

    ChannelFactoryInitialize(0, args.iface)
    client = LocoClient()
    client.SetTimeout(10.0)
    if hasattr(client, "Init"):
        client.Init()

    fsm = client.GetFsmId()
    print(f"当前 FSM: {fsm}")
    if fsm != 500:
        print("WARN: 当前不是 FSM 500，移动可能失败。建议先运行 prepare_walk.py。")
        confirm("仍要继续吗？")

    confirm(
        f"将发送: vx={args.vx}, vy={args.vy}, omega={args.omega}，"
        f"持续 {args.duration}s。\n"
        "确认周围清空、急停在手。"
    )

    print("发送速度...")
    ret = client.SetVelocity(args.vx, args.vy, args.omega, args.duration)
    print(f"SetVelocity 返回: {ret}  (0 = 成功)")

    time.sleep(args.duration + 0.1)
    stop_ret = client.SetVelocity(0.0, 0.0, 0.0, 0.5)
    print(f"停止指令返回: {stop_ret}")

if __name__ == "__main__":
    main()
